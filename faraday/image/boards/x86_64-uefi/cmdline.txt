# Faraday: the board command line, with faraday.dpi=160 and freed and allocated memory zeroed by the kernel
# (docs/history/PLAN.md §4.2), so a locked process's memory cannot outlive it.
#
# usbcore.authorized_default=2 authorises only devices on hard-wired
# internal ports; every other USB device waits for faraday-grant, which
# authorises its interfaces one class at a time (docs/history/PLAN.md §4.6).
#
# fbcon=logo-pos:center: a PC's boot logo is a 320 x 320 square (its
# panel's size is not known when the image is built), which the kernel
# would draw at the top left of the screen; this puts it in the middle.
#
# loglevel=5, after common's `quiet`: the framebuffer console draws no
# logo when the console log level is `quiet`'s 4 or lower (fbcon_init in
# drivers/video/fbdev/core/fbcon.c), so with `quiet` alone a PC shows a
# blank screen until the app's first frame. `quiet` stays because the EFI
# stub reads only that word, and without it prints its own lines on the
# screen before the kernel starts. The release image names no console, so
# the higher level prints nothing anywhere; the dev variant removes it.
#
# random.trust_cpu=off random.trust_bootloader=off: neither the CPU's
# RDRAND nor the firmware's seed is credited as entropy (they are still
# mixed in). Linux 6.2 removed the Kconfig options for this; these two
# parameters are all that is left, and check-kernel-config.sh checks
# that CONFIG_CMDLINE carries both (docs/OWN-RANDOMNESS.md §3.3).
faraday.dpi=160 init_on_free=1 init_on_alloc=1 usbcore.authorized_default=2
random.trust_cpu=off random.trust_bootloader=off
fbcon=logo-pos:center
loglevel=5
