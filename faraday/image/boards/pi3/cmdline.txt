# The Raspberry Pi's part of the kernel command line.

# The console is the PL011 on the GPIO header, which disable-bt in
# config.txt frees for serial0, and nothing else: no kernel text can land
# on the panel.
console=serial0,115200

# Faraday: freed and allocated memory zeroed by the kernel (docs/SECURITY-MODEL.md "What the kernel can reach"),
# and only hard-wired USB devices authorised without faraday-grant (§4.6).
init_on_free=1
init_on_alloc=1
usbcore.authorized_default=2

# Faraday: no CPU generator and no bootloader seed credited, as on the
# stick (docs/OWN-RANDOMNESS.md §3.2, §3.3). Linux 6.2 removed the Kconfig
# options and trusts both by default. On 32-bit ARM trust_cpu=off has
# nothing to act on, as there is no CPU generator, and trust_bootloader=off
# comes too late for the firmware's seed, which the kernel reads before it
# parses the command line; the Pi's kernel patch 0004 (docs/KERNEL-PATCHES.md)
# makes the seed uncounted by default. Both parameters are kept for the
# kernel's other paths.
random.trust_cpu=off
random.trust_bootloader=off

# Faraday: the SoC's hardware generator, bcm2835-rng, is mixed into the pool
# and credited with nothing (docs/OWN-RANDOMNESS.md §3.5). In the Raspberry
# Pi 6.18 tree, hwrng_register in drivers/char/hw_random/core.c sets the
# driver's quality to min(default_quality, 1024, its own or 1024), and
# bcm2835-rng states none; default_quality is the built-in `rng_core`
# module's parameter. At 0 the hwrng thread (hwrng_fillfn) still runs and
# calls add_hwgenerator_randomness with a credit of 0: mixed, not counted.
rng_core.default_quality=0

# Faraday: loglevel=5, after common's `quiet`. The framebuffer console
# draws no logo when the console log level is `quiet`'s 4 or lower
# (fbcon_init in drivers/video/fbdev/core/fbcon.c), so with `quiet` alone
# the panel stays blank until the app's first frame. The only console is
# the serial port, which now also carries warnings; the dev variant
# removes it.
loglevel=5

# The camera driver (built in) takes any frame above 1280 x 720 through its
# stills path, which is slow; the scanner asks for 1920 x 1080 so that a
# dense QR has pixels enough to read, and these raise the driver's video
# limits to it.
bcm2835-v4l2.max_video_width=1920
bcm2835-v4l2.max_video_height=1080

# An HDMI screen larger than 11 inches gives no density the shell believes,
# so dp are pixels, as on the stick (docs/ORIENTATION.md §2.1); a panel's
# values from the boot script (--dpi) come first.
faraday.dpi=160

# The boot logo is a small square, since the screen's size is not known
# when the image is built; this puts it in the middle of the screen rather
# than at its top left.
fbcon=logo-pos:center
