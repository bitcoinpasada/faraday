# The Raspberry Pi's part of the kernel command line.

# The console is the PL011 on the GPIO header, which disable-bt in
# config.txt frees for serial0, and nothing else: no kernel text can land
# on the panel.
console=serial0,115200

# Faraday: freed and allocated memory zeroed by the kernel (docs/history/PLAN.md §4.2),
# and only hard-wired USB devices authorised without faraday-grant (§4.6).
init_on_free=1
init_on_alloc=1
usbcore.authorized_default=2

# Faraday: loglevel=5, after common's `quiet`. The framebuffer console
# draws no logo when the console log level is `quiet`'s 4 or lower
# (fbcon_init in drivers/video/fbdev/core/fbcon.c), so with `quiet` alone
# the panel stays blank until the app's first frame. The only console is
# the serial port, which now also carries warnings; the dev variant
# removes it.
loglevel=5

# An HDMI screen larger than 11 inches gives no density the shell believes,
# so dp are pixels, as on the stick (docs/ORIENTATION.md §2.1); a panel's
# values from the boot script (--dpi) come first.
faraday.dpi=160

# The boot logo is a small square, since the screen's size is not known
# when the image is built; this puts it in the middle of the screen rather
# than at its top left.
fbcon=logo-pos:center
