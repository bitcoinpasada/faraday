# Faraday: upstream's pi3 dev/cmdline.txt with -loglevel=5 added.
#
# The dev variant on a Raspberry Pi: the panel gets a console too, so a
# board with no serial adapter still shows the probe lines.
console=tty1

# Faraday: the release board's loglevel=5 is for the logo; the dev variant
# prints every level, as its `-quiet` intends, until the app starts (the dev
# inittab then lowers the level).
-loglevel=5

# Faraday: the dev inittab sends the app's error output to /dev/kmsg, which
# limits how many lines userspace writes there; on the dev variant every one
# goes through.
printk.devkmsg=on

# Faraday: draw in a centred 1366x768 window of a larger screen, the rest black
# (docs/FRAMES.md section 6). A screen that cannot hold it is drawn whole.
# Take this word off the card's cmdline.txt to time the whole screen.
faraday.window=1366x768
